import numpy as np
import matplotlib.pyplot as plt

hours = [1, 2, 3, 4]
scores = [50, 60, 70, 80]

r = np.corrcoef(hours, scores)[0, 1]
print("Pearson Correlation Coefficient:", r)

plt.scatter(hours, scores)
plt.title("Hours Studied vs Test Scores")
plt.xlabel("Hours")
plt.ylabel("Scores")
plt.show()
